[[steps]]
heading = "Rules, then the verdict"
text = "The **Hard Rules** are checked as you type, so a broken one shows before you save. Then **Professor Aurel** reads your sentence and gives a verdict on the **Soft Rules**: followed the plan or not. Disagree? **Dispute** it, explaining why in a sentence."
why = "The verdict is about the process, not the money. A trade that followed every rule and lost is marked as following the plan."

[[steps]]
heading = "Close it, and log skips too"
text = "When the trade closes, add the **exit price**, an **Emotion Tag** (calm, FOMO, revenge, fear, overconfident) and **one lesson** in a line. Setups you passed on go in as **skipped setups**: when, which way and why. They show your plan working."
why = "Skipped setups don't count towards Live Readiness, but they appear in your Weekly Review, and a week of good skips is a good week."

[[steps]]
heading = "The Weekly Review"
text = "Once a week, open the **Weekly Review**: trades taken and skipped, how many broke the plan, win rate, **expectancy** in R, which rules broke, how you felt. Then write **one thing to keep doing and one to stop**. Rules only change here, never mid-trade."
why = "CME Group recommends a daily post-mortem where the why and how of your trading matter more than adding up profit or loss. The weekly note turns that into one concrete change at a time."

[[steps]]
heading = "The road to Live Readiness"
text = "**Live Readiness** needs four things at once: at least **50 Demo Trades**, over at least **8 weeks**, with at least **90%** following the plan, and no breach of your plan's **max total loss**. Profit isn't required. Discipline is."
why = "It unlocks the Live Track and the Advanced Track. Until then, every trade is practice for the habit, not a test of luck."
