[[steps]]
heading = "1. Trend first, on H4"
text = "Read the **H4 trend from structure**: higher highs and higher lows for an uptrend, the reverse for a downtrend (Lesson 25). No clear trend, no setup today. The trade only goes in the direction H4 is already going."
why = "This is Lesson 22's top-down rule: the higher timeframe decides the direction, the lower one finds the entry."

[[steps]]
heading = "2. A zone you wrote down"
text = "On **H1**, wait for price to pull back into a **support or resistance zone** you named in that day's **Scenarios** (Lesson 38). A Fibonacci level in the zone adds confluence (Lesson 31); it's welcome, not required."
why = "Writing the zone before the session matters: a zone you notice only once price is there is often one you've talked yourself into. Your plan's Soft Rule asks that every trade be one of the day's Scenarios."

[[steps]]
heading = "3. A confirmation candle closes"
text = "At the zone, wait for a **confirmation candle** in the trend's direction, a pin bar or an engulfing, to **close** (Lesson 26). Price touching the zone isn't a signal. The close is."
why = "CME Group's trade-plan course stresses clear entry criteria that you stick to, because otherwise decisions drift towards emotion rather than strategy."

[[steps]]
heading = "4. Stop, target, size"
text = "**Stop** about 1–1.5 ATR beyond the candle's wick and the zone, off round numbers (Lesson 32). **Target** the next zone: at least **2:1**, or skip. **Size** with the calculator so a loss is at most 1%, rounded down (Lesson 33)."
why = "The order matters: stop from the chart, then target, then size. Never the other way round."

[[steps]]
heading = "5. Last checks, then click"
text = "Before the order goes in: no **No-Trade Window** within 5 minutes, the **Daily Loss Limit** not reached, under your **trades-per-day** cap. Any no means **skip**, and log it as a skipped setup. Then place the order with its stop and target attached."
why = "Your journal checks these Hard Rules as you type, but the habit is to check them before you click, not after."
