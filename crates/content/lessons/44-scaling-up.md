[[steps]]
heading = "Earn the right to scale"
text = "Raise your risk only after **three live months** with at least **90%** of trades following your plan. Profit is welcome, but **adherence** is the test: it shows your discipline holds with real money and real nerves, over enough trades to mean something."
why = "This mirrors Live Readiness: discipline first, measured over time. Three months live usually means several dozen trades, enough to see habits rather than luck."

[[steps]]
heading = "Small steps, 0.25% at a time"
text = "Step up by **0.25%** per trade: 0.5% → 0.75% → 1%. Save each step as a **new plan version**. Hold each level for at least a **month at 90%+** before the next. If your adherence drops at a new level, go back one step."
why = "A small step lets you notice whether bigger amounts change your behaviour before they can do real damage."

[[steps]]
heading = "1% stays the ceiling"
text = "This course never goes above **1%** per trade. As the account grows, the same 1% is more money, so your size **grows on its own**. Scaling up means reaching 1% carefully, not going past it."
why = "CME Group's tables show why fixed small percentages matter: at 2% a five-loss streak takes $50,000 to $45,198, at 1% only to $47,550, and deeper losses need ever bigger gains to recover (Lesson 35)."

[[steps]]
heading = "Step down after a drawdown"
text = "If the account falls about **10% below its peak**, step **back down one level** and review your journal for leaks (Lesson 41). Only scale up again after a new high and another month at 90%+. Size up from strength, never to win losses back."
why = "Cutting risk in a drawdown keeps the hole shallow, where recovery is realistic: −10% needs +11%, but −30% needs about +43%."
