[[steps]]
heading = "Support et résistance"
text = "Le **support** est une zone de prix où un prix qui baisse a tendance à s’arrêter ou à remonter. La **résistance**, une zone où un prix qui monte a tendance à s’arrêter ou à redescendre. Les leçons de CME Group les décrivent comme des niveaux où le prix a **plus de chances** de faire une pause ou de changer de direction."
why = "Ils viennent de la mémoire : les traders se souviennent des endroits où le prix a tourné et y agissent à nouveau. Le support est en général sous le prix actuel, la résistance au-dessus."

[[steps]]
heading = "Des zones, pas des lignes"
text = "Le prix tourne rarement au même centime. Trace une **zone** de quelques dollars autour des retournements, qui couvre les mèches et les corps. Avec une seule ligne exacte, tu croirais qu’un niveau a cédé alors que le prix l’a juste dépassé d’un cheveu."
why = "Ce cours commence avec des zones de quelques dollars en H4. Si une zone doit faire 20 $ de large pour contenir tous les retournements, ce sont sans doute deux niveaux, ou aucun."

[[steps]]
heading = "Fiable après deux réactions"
text = "Un niveau mérite confiance quand le prix y a **réagi au moins deux fois** : il a tourné là, est parti, est revenu et a tourné à nouveau. Un seul contact, ce n’est qu’un virage. Plus il y a de réactions nettes, plus les traders surveillent ce niveau."
why = "Deux réactions, c’est la règle pratique de ce cours, pas une loi du marché. La leçon 28 trade exactement ça : un plancher ou un plafond que le prix a déjà respecté deux fois."

[[steps]]
heading = "Les chiffres ronds"
text = "Les prix comme **4 100,00** ou **4 050,00** servent souvent de niveaux eux aussi, parce que beaucoup de traders y placent des ordres. CME Group montre le pétrole brut incapable de franchir 100 $ pendant des années. Traite les paliers de 50 $ et de 100 $ de l’or comme des endroits à surveiller, pas comme des murs."
why = "Un chiffre rond qui est aussi un ancien point de retournement est plus solide que chacun des deux seul. L’étape 6 du cours appelle ça la confluence."

[[steps]]
heading = "Les niveaux cassés changent de rôle"
text = "Quand le prix casse un support et reste en dessous, cette zone devient souvent une **résistance** quand il remonte. Une résistance cassée devient souvent un support. CME Group décrit cette inversion des rôles : c’est pour ça qu’un niveau cassé reste sur ton graphique."
why = "Le prix revient souvent tester un niveau cassé par l’autre côté avant de repartir. Ce retest est un setup à part entière dans le parcours avancé (cassure et retest)."

[[steps]]
heading = "Trade la réaction, pas le niveau"
text = "Une zone te dit **où** faire attention, pas quoi faire. N’agis que sur une réaction : le prix qui ralentit, une bougie de confirmation, dans le sens de l’unité de temps supérieure, avec un stop au-delà de la zone."
why = "Les niveaux cassent régulièrement. Le stop au-delà de la zone transforme un niveau qui cède en une petite perte prévue, au lieu d’un espoir."
