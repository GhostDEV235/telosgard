[[steps]]
heading = "Un niveau testé deux fois"
text = "Le jeu du rebond trade une zone sur laquelle le prix a **réagi au moins deux fois** (leçon 24). Un plancher testé deux fois, c’est un **double creux** ; un plafond, un **double sommet**. CME Group les décrit comme un marché qui échoue deux fois à dépasser le même prix."
why = "La leçon de CME Group sur les retournements : à sa deuxième tentative, le marché n’a pas pu faire un creux plus bas (ou un sommet plus haut), ce qui suggère qu’il manque peut-être d’élan dans ce sens."

[[steps]]
heading = "Où il convient"
text = "Il fonctionne le mieux dans un **range** (leçon 25), entre plancher et plafond, ou sur une zone majeure dans le sens de l’unité supérieure. Acheter un rebond sur un support dans une tendance baissière en H4, c’est aller contre la tendance : moins de chances, donc les débutants y renoncent."
why = "Les bords d’un range sont le terrain du jeu du rebond. Dans une forte tendance, le jeu du repli convient en général mieux."

[[steps]]
heading = "La confirmation, puis le trade"
text = "Même discipline que pour le repli : attends qu’une **bougie de confirmation** **clôture** dans la zone, place le stop au-delà de sa mèche et de la zone, et vise la zone suivante, souvent le milieu ou l’autre bord du range. Au moins deux fois le risque, sinon renonce."
why = "Dans un range, un premier objectif raisonnable est le milieu : l’autre bord est la résistance suivante, et le prix cale souvent à mi-chemin."

[[steps]]
heading = "Quand les niveaux cassent"
text = "Les niveaux ne tiennent pas éternellement : chaque test consomme une partie des ordres qui attendaient là. Si une bougie **clôture au-delà** de la zone, l’idée du rebond est morte. Le stop te sort avec une perte prévue, et le niveau cassé change peut-être de rôle (leçon 24)."
why = "C’est pour ça que le stop va au-delà de la zone, pas à l’intérieur : si le prix ne fait que l’effleurer et revient, tu restes ; s’il clôture au-delà, tu sors."

[[steps]]
heading = "Repli ou rebond ?"
text = "Le **repli** trade avec une tendance après une pause. Le **rebond** trade sur un niveau éprouvé, souvent dans un range. Les deux relèvent du Price Action classique : une bougie de confirmation sur un niveau, avec la tendance ou dans le range, et un stop."
why = "Ton Plan de trading précise quels jeux tu prends. L’étape 8 du cours le met par écrit ; l’étape 6 ajoute Fibonacci et la confluence aux deux."
