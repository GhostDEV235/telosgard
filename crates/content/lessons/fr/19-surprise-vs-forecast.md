[[steps]]
heading = "Précédent, prévision, réel"
text = "Chaque publication du calendrier affiche trois chiffres : le **précédent** (la dernière fois), la **prévision** (ce qu’attendent les économistes) et le **réel** (ce qui est sorti). Les marchés ont déjà intégré la prévision dans les prix. Ce qui fait bouger le prix, c’est la **surprise** : le réel comparé à la prévision."
why = "La prévision est en général le milieu des estimations de nombreux économistes, recueillies avant la publication. Les calendriers l’appellent parfois le consensus."

[[steps]]
heading = "Chaude ou molle ?"
text = "Une statistique **chaude** signale une économie plus forte ou une inflation plus élevée que prévu : CPI au-dessus de la prévision, plus d’emplois que prévu. Une statistique **molle**, c’est l’inverse. Attention au chômage : un taux **plus élevé** que prévu est **mou**, parce qu’il signale un marché de l’emploi plus faible."
widget = "surprise_meter"
why = "Conforme veut dire proche de la prévision ; les projets du marché changent à peine. La taille d’écart qui compte comme surprise dépend de la statistique ; ce cours reste simple : au-dessus, en dessous ou conforme."

[[steps]]
heading = "Dans quel sens va l’or, en général"
text = "Une statistique chaude fait attendre des taux de la Fed plus élevés : les rendements et le dollar ont tendance à monter, et l’or **baisse en général**. Une statistique molle laisse attendre des taux plus bas : l’or **monte en général**. C’est l’enchaînement des leçons 16 et 17."
why = "L’enchaînement : statistique chaude → on attend des taux plus élevés → rendements plus hauts et dollar plus fort → coût d’opportunité de l’or plus élevé → or en baisse. Une statistique molle fait tout le chemin à l’envers."

[[steps]]
heading = "Pourquoi ça peut partir dans l’autre sens"
text = "Le chiffre principal peut être chaud alors que le détail est mou, le chiffre du mois précédent peut être révisé, ou la surprise peut être plus petite que ce que les traders craignaient. L’or peut alors faire l’**inverse** de la réaction attendue, parfois en quelques minutes."
why = "C’est pour ça que ce cours enseigne la réaction comme une tendance à reconnaître, pas comme un trade à prendre. Tu en verras davantage dans la leçon « La suite d’une annonce » du Parcours avancé, après « Prêt pour le réel »."

[[steps]]
heading = "Ne trade pas la bougie de la publication"
text = "Dans les premières secondes après une grande statistique, les spreads s’élargissent, les stops glissent et le prix fait souvent des allers-retours. Ta **Fenêtre sans trade** te tient à l’écart 5 minutes de chaque côté. Ensuite, prends seulement ta configuration habituelle sur un niveau, avec un stop."
why = "Trader la publication, c’est parier sur un sens avec les pires coûts de la journée. Attendre ne te coûte rien ; la configuration de ton plan est toujours là après."
