[[steps]]
heading = "Mériter de grossir"
text = "N’augmente ton risque qu’après **trois mois en réel** avec au moins **90 %** des trades conformes au plan. Le profit est bienvenu, mais c’est le **respect du plan** qui compte : il montre que ta discipline tient avec de l’argent réel et de vrais nerfs, sur assez de trades pour vouloir dire quelque chose."
why = "C’est le même principe que « Prêt pour le réel » : la discipline d’abord, mesurée dans le temps. Trois mois en réel, c’est en général plusieurs dizaines de trades, assez pour voir des habitudes plutôt que de la chance."

[[steps]]
heading = "Des petites marches de 0,25 %"
text = "Monte de **0,25 %** par trade : 0,5 % → 0,75 % → 1 %. Enregistre chaque marche dans une **nouvelle version du plan**. Tiens chaque niveau au moins **un mois à 90 % ou plus** avant le suivant. Si ton respect du plan baisse à un nouveau niveau, redescends d’une marche."
why = "Une petite marche te laisse voir si des montants plus gros changent ton comportement avant qu’ils ne puissent faire de vrais dégâts."

[[steps]]
heading = "1 % reste le plafond"
text = "Ce cours ne dépasse jamais **1 %** par trade. Quand le compte grandit, le même 1 % représente plus d’argent : ta taille **grandit toute seule**. Grossir, c’est atteindre 1 % avec prudence, pas le dépasser."
why = "Les tableaux de CME Group montrent pourquoi de petits pourcentages fixes comptent : à 2 %, une série de cinq pertes fait passer 50 000 $ à 45 198 $, à 1 % seulement à 47 550 $, et plus la perte est profonde, plus le gain nécessaire pour revenir est grand (leçon 35)."

[[steps]]
heading = "Redescendre après un drawdown"
text = "Si le compte tombe d’environ **10 % sous son sommet**, **redescends d’une marche** et cherche les fuites dans ton journal (leçon 41). Ne remonte qu’après un nouveau plus haut et un mois de plus à 90 % ou plus. On grossit en position de force, jamais pour se refaire."
why = "Baisser le risque pendant un drawdown garde le trou peu profond, là où se refaire reste réaliste : −10 % demande +11 %, mais −30 % demande environ +43 %."
